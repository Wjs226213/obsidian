## 常用函数速查

### OpenCV

```python
# 色彩空间转换
cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# 图像调整
cv2.resize(img, (width, height))
img[y1:y2, x1:x2]  # 裁剪

# 图像模糊
cv2.blur(img, (kernel, kernel))
cv2.GaussianBlur(img, (kernel, kernel), 0)
cv2.medianBlur(img, kernel)

# 阈值处理
cv2.threshold(gray, thresh, maxval, type)
cv2.adaptiveThreshold(gray, maxval, method, type, blockSize, C)

# 图像绘制
cv2.rectangle(img, pt1, pt2, color, thickness)
cv2.circle(img, center, radius, color, thickness)
cv2.putText(img, text, org, fontFace, fontScale, color, thickness)

# 边缘检测
cv2.Canny(img, threshold1, threshold2)
cv2.dilate(img, kernel)
cv2.erode(img, kernel)

# 轮廓处理
cv2.findContours(image, mode, method)
cv2.drawContours(image, contours, contourIdx, color, thickness)
cv2.contourArea(contour)
cv2.boundingRect(contour)
```
